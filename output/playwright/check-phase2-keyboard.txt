async(page)=>{await page.getByRole("button",{name:"Exit Focus",exact:true}).focus();await page.keyboard.press("Tab");const geometry=await page.evaluate(()=>{const e=document.querySelector(".kami-mode-controls"),b=document.activeElement.getBoundingClientRect(),r=e.getBoundingClientRect();return {active:document.activeElement.getAttribute("aria-label"),width:e.clientWidth,scroll:e.scrollLeft,leafScroll:app.workspace.activeLeaf.view.containerEl.scrollLeft,fullyVisible:b.left>=r.left-1&&b.right<=r.right+1}});if(!geometry.fullyVisible||geometry.leafScroll!==0)throw Error(JSON.stringify(geometry));await page.screenshot({path:"/tmp/kami-phase2-narrow.png"});await page.keyboard.press("Enter");const paperExit=await page.evaluate(()=>!app.workspace.activeLeaf.view.containerEl.classList.contains("kami-white-page-preview-active"));await page.evaluate(()=>window.phase2MenuFor(app.workspace.activeLeaf.view));await page.keyboard.press("Escape");const preserved=await page.evaluate(()=>document.body.classList.contains("kami-focus-open"));await page.keyboard.press("Escape");const exited=await page.evaluate(()=>!document.body.classList.contains("kami-focus-open"));if(!paperExit||!preserved||!exited)throw Error(JSON.stringify({paperExit,preserved,exited}));return {geometry,paperExit,menuEscapePreservesFocus:preserved,nextEscapeExits:exited};}
